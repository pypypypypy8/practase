
#include <iostream>
using namespace std;

int main()

long long first = 0;
    long long last = 0;

    if (!(cin >> first >> last))
    {
        cerr << "Error: expected two integers\n";
        return 1;
    }

    if (first > last)
    {
        swap(first, last);
    }

    long long sum = 0;
    for (long long i = first; i <= last; ++i)
    {
        sum += i;
    }

   cout << "sum=" << sum << '\n';
    return 0;
}
