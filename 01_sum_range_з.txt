#include <algorithm>
#include <iostream>

int main()
{
    long long first = 0;
    long long last = 0;

    if (!(std::cin >> first >> last))
    {
        std::cerr << "Error: expected two integers\n";
        return 1;
    }

    if (first > last)
    {
        std::swap(first, last);
    }

    long long sum = 0;
    for (long long i = first; i <= last; ++i)
    {
        sum += i;
    }

    std::cout << "sum=" << sum << '\n';
    return 0;
}
